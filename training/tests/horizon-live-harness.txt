  timeoutMs=15000; busy=false; failure:string|null=null; seed:number|null=1; steps=0; maxSteps=64; scenario='standard';

  effect: ((input:GameAction)=>void)|null=null;
  game:any;
  constructor() {
    this.body.actionProcessor={process:(a:GameAction)=>{
      this.calls.push({...a});
      if(this.effect){this.effect(a);return;}
      this.count++; this.body.turnCount++;
      if(a.type==='Directional'){this.body.x=a.targetX;this.body.y=a.targetY;}
      if(a.type==='DismissInteraction'||a.type==='LadderCancel')this.decision='world';
    }};
    this.game={body:this.body,replayManager:{getStats:()=>({count:this.count})}};
  }
  private player(): {health:number;getRoom:()=>RoomFixture;[key:string]:any} {return this.body;}
